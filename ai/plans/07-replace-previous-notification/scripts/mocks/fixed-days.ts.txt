import { formatPrayerTime } from '@/shared/time';
import type { IApiResponse } from '@/shared/types';

/**
 * Fixed-date mock days for session 7's notification proof: the clock is driven BACK to these dates
 * (a backward jump fires no alarm), so the days must not move with the download. The six times of
 * every day are real London readings copied verbatim from mocks/simple.ts's fixed blocks (the 2024
 * spring drift set), exactly as that file copies them. Jamat fields are unused placeholders, as in
 * every mock.
 *
 * On 2026-09-13 the rows the proof uses: Suhoor 04:22 (derived, 20 minutes before Fajr 04:42),
 * Fajr 04:42, Asr 16:40.
 */

const MINUTE = 60_000;

/** Kept because build-mock.zsh requires it; these fixed days never need it */
export const addMinutes = (from: Date, minutesToAdd: number) =>
  formatPrayerTime(new Date(from.getTime() + minutesToAdd * MINUTE));

const day = (date: string, fajr: string, sunrise: string, dhuhr: string, asr: string, magrib: string, isha: string) => ({
  date,
  fajr,
  sunrise,
  dhuhr,
  asr,
  magrib,
  isha,
  fajr_jamat: '00:00',
  dhuhr_jamat: '00:00',
  asr_2: '00:00',
  asr_jamat: '00:00',
  magrib_jamat: '00:00',
  isha_jamat: '00:00',
});
