/**
 * The tile cache: what it keeps, and how it stays under the owner's 25 MB cap
 */

import { cachedBytes, clearTiles, readTile, TILE_CACHE_CAP_BYTES, writeTile } from '@/shared/tileCache';
import { database } from '@/stores/database';

const MEGABYTE = 1024 * 1024;

/** A tile of a given size, filled with a recognisable byte so a mixed-up read is visible */
const tileOf = (megabytes: number, fill: number): Uint8Array => new Uint8Array(megabytes * MEGABYTE).fill(fill);

describe('the tile cache', () => {
  // This suite writes raw MMKV rather than atoms, which the harness does not reset for us
  beforeEach(() => {
    database.clearAll();
  });

  it('keeps a tile it was given', () => {
    const bytes = tileOf(1, 7);

    writeTile('london-0-0', bytes);

    expect(readTile('london-0-0')).toEqual(bytes);
  });

  it('reports nothing for a tile it has never seen', () => {
    expect(readTile('never-fetched')).toBeNull();
  });

  it('keeps two locations at once', () => {
    writeTile('london-0-0', tileOf(1, 1));
    writeTile('jakarta-0-0', tileOf(1, 2));

    expect(readTile('london-0-0')?.[0]).toBe(1);
    expect(readTile('jakarta-0-0')?.[0]).toBe(2);
  });

  it('reports how many bytes it is holding', () => {
    writeTile('a', tileOf(1, 1));
    writeTile('b', tileOf(1, 2));
    writeTile('c', tileOf(1, 3));

    expect(cachedBytes()).toBe(3 * MEGABYTE);
  });

  it('evicts the least recently used tile when the cap is passed', () => {
    writeTile('first', tileOf(10, 1));
    writeTile('second', tileOf(10, 2));
    readTile('first');

    writeTile('third', tileOf(10, 3));

    expect(readTile('second')).toBeNull();
    expect(readTile('first')).not.toBeNull();
    expect(readTile('third')).not.toBeNull();
  });

  it('never exceeds the cap after an eviction', () => {
    writeTile('a', tileOf(10, 1));
    writeTile('b', tileOf(10, 2));
    writeTile('c', tileOf(10, 3));
    writeTile('d', tileOf(10, 4));

    expect(cachedBytes()).toBeLessThanOrEqual(TILE_CACHE_CAP_BYTES);
  });

  it('evicts more than one tile when a single tile needs the room', () => {
    for (let index = 0; index < 12; index += 1) writeTile(`filler-${index}`, tileOf(2, index));

    writeTile('big', tileOf(6, 99));

    expect(cachedBytes()).toBeLessThanOrEqual(TILE_CACHE_CAP_BYTES);
    expect(readTile('big')).not.toBeNull();
  });

  it('refuses a tile larger than the whole cap', () => {
    writeTile('keeper', tileOf(1, 1));

    writeTile('enormous', tileOf(30, 9));

    expect(readTile('enormous')).toBeNull();
    expect(readTile('keeper')).not.toBeNull();
  });

  it('treats a read as a use, so the tile a user keeps opening survives', () => {
    writeTile('older', tileOf(10, 1));
    writeTile('newer', tileOf(10, 2));

    readTile('older');
    writeTile('newest', tileOf(10, 3));

    expect(readTile('older')).not.toBeNull();
    expect(readTile('newer')).toBeNull();
  });

  it('forgets everything when asked', () => {
    writeTile('a', tileOf(1, 1));
    writeTile('b', tileOf(1, 2));

    clearTiles();

    expect(readTile('a')).toBeNull();
    expect(cachedBytes()).toBe(0);
  });

  it('caps at the 25 MB the owner asked for', () => {
    expect(TILE_CACHE_CAP_BYTES).toBe(25 * 1024 * 1024);
  });
});
