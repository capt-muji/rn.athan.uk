import { formatPrayerTime } from '@/shared/time';
import type { IApiResponse } from '@/shared/types';

/**
 * Fixed-date mock days for session 9's proof of finding 74: the clock is driven to Friday 25
 * September 2026, whose Magrib 00:40 and Isha 01:30 fall on Saturday morning and belong to Friday's
 * list. The five days are the brief's table verbatim (session 1's high-latitude mock); every time is
 * a well-formed HH:mm the client's validation reads. Jamat fields are unused placeholders, as in
 * every mock.
 *
 * No surrounding days are needed: the proof never drives past Saturday 01:35, and the widest window
 * the app can read there (the Extras night rows' three list days) reaches only 2026-09-28.
 */

const MINUTE = 60_000;

/** Kept because build-mock.zsh requires it; these fixed days never need it */
export const addMinutes = (from: Date, minutesToAdd: number) =>
  formatPrayerTime(new Date(from.getTime() + minutesToAdd * MINUTE));

const day = (date: string, fajr: string, sunrise: string, dhuhr: string, asr: string, magrib: string, isha: string) => ({
  date,
  fajr,
  sunrise,
  dhuhr,
  asr,
  magrib,
  isha,
  fajr_jamat: '00:00',
  dhuhr_jamat: '00:00',
  asr_2: '00:00',
  asr_jamat: '00:00',
  magrib_jamat: '00:00',
  isha_jamat: '00:00',
});
