/**
 * Ten calibration animations for the qibla sheet, all in the app's own qibla palette.
 *
 * The brief: 🐋  "Really ugly. Super ugly. Okay, scrap it, redesign it from the ground up... design 10
 * different animations... they should look really, really illustrative because a lot of people will not
 * read and don't know how to read... people are visual learners."
 *
 * So every design teaches the gesture with no words at all, and looks like it belongs on this sheet.
 * References consulted: Apple's own compass HUD (tilt to roll a ball around a ring until it FILLS, a
 * progress metaphor rather than a bare instruction), Android's figure-eight prompt, and the app's own
 * Rub el Hizb jewel from the compass design the owner locked in.
 *
 * Revised after reading every strip back: the first pass drew everything too small for a sheet the user
 * holds at arm's length, the stars were spiky rather than Islamic, the arch read as a rounded rectangle
 * rather than architecture, the ripples were too faint to see, and design 8's ticks overflowed the cell
 * while its jewel collided with the phone.
 */

import { P, s } from './palette.mjs';

const TURN = Math.PI * 2;
const r2 = (v) => Number(v.toFixed(2));

/** The lemniscate the gesture traces, as the app's own `wavePoint` computes it */
const fig8 = (t, w, h) => {
  const a = t * TURN;
  return { x: (w / 2) * Math.sin(a), y: h * Math.sin(a) * Math.cos(a) };
};

const fig8Path = (w, h, n = 72) => {
  const pts = Array.from({ length: n }, (_, i) => fig8(i / n, w, h));
  return `${pts.map((p, i) => `${i === 0 ? 'M' : 'L'}${r2(p.x)},${r2(p.y)}`).join('')}Z`;
};

/** The lean off the curve's own tangent, which is what teaches a wrist that rolls */
const lean = (t, deg = 22) => {
  const a = t * TURN;
  return Math.sin(Math.atan2(2 * Math.cos(2 * a), Math.cos(a))) * deg;
};

/** A phone glyph centred on its own origin, so one transform places it */
const phone = (w, h, { stroke = P.accent, sw = 2.4, glow = 0 } = {}) => {
  const rad = w * 0.2;
  return `${glow ? `<rect x="${-w / 2 - glow}" y="${-h / 2 - glow}" width="${w + glow * 2}" height="${h + glow * 2}" rx="${rad + glow}" fill="${P.accent}" opacity="0.12"/>` : ''}
<rect x="${-w / 2}" y="${-h / 2}" width="${w}" height="${h}" rx="${rad}" fill="${P.kaaba}" stroke="${stroke}" stroke-width="${sw}"/>
<rect x="${-w / 2 + 4}" y="${-h / 2 + 5}" width="${w - 8}" height="${h - 10}" rx="${rad * 0.55}" fill="${s(0.2)}"/>
<rect x="-3" y="${-h / 2 + 2}" width="6" height="1.8" rx="0.9" fill="${s(0.45)}"/>`;
};

/** The app's Rub el Hizb jewel: two squares at 45 degrees, outline only */
const jewel = (rad, { stroke = P.accent, sw = 1.8, op = 1 } = {}) =>
  `<g opacity="${op}" fill="none" stroke="${stroke}" stroke-width="${sw}" stroke-linejoin="round">
<rect x="${-rad}" y="${-rad}" width="${rad * 2}" height="${rad * 2}"/>
<rect x="${-rad}" y="${-rad}" width="${rad * 2}" height="${rad * 2}" transform="rotate(45)"/></g>`;

/** An eight-point Islamic star with SOFT points, drawn from two overlaid squares rather than spikes */
const star8 = (rad, { fill = P.accent, op = 1, stroke = 'none' } = {}) => {
  const pts = Array.from({ length: 16 }, (_, i) => {
    const a = (i / 16) * TURN - Math.PI / 2;
    const rr = i % 2 ? rad * 0.72 : rad;
    return `${r2(Math.cos(a) * rr)},${r2(Math.sin(a) * rr)}`;
  });
  return `<polygon points="${pts.join(' ')}" fill="${fill}" opacity="${op}" stroke="${stroke}" stroke-width="1.4" stroke-linejoin="round"/>`;
};
