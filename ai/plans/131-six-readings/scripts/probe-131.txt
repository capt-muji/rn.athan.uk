import { useEffect, useRef, useState } from 'react';
import { Platform, SafeAreaView, StyleSheet, Text } from 'react-native';

import { getLastKnownPositionAsync } from 'expo-location';
import * as Next from 'expo-location/next';

type NextFix = {
  timestamp: number;
  horizontalAccuracy: number | null;
  heading: number | null;
  headingAccuracy: number | null;
  speed: number | null;
  mocked: boolean;
};

type ClassicFix = {
  timestamp: number;
  accuracy: number | null;
  heading: number | null;
  speed: number | null;
};

const log = (marker: string, label: string, value?: unknown) => {
  const body =
    value === undefined ? '' : typeof value === 'string' ? value : JSON.stringify(value);
  console.log(`[RD${marker}] ${label}${body ? ` ${body}` : ''}`);
};

const nextFixOf = (
  position: NextFix | null,
): NextFix | null =>
  position === null
    ? null
    : {
        timestamp: position.timestamp,
        horizontalAccuracy: position.horizontalAccuracy,
        heading: position.heading,
        headingAccuracy: position.headingAccuracy,
        speed: position.speed,
        mocked: position.mocked,
      };

const classicFixOf = (
  position: { timestamp: number; coords: { accuracy?: number | null; heading?: number | null; speed?: number | null } } | null,
): ClassicFix | null =>
  position === null
    ? null
    : {
        timestamp: position.timestamp,
        accuracy: position.coords.accuracy ?? null,
        heading: position.coords.heading ?? null,
        speed: position.coords.speed ?? null,
      };

const wait = (ms: number) => new Promise<void>((resolve) => { setTimeout(resolve, ms); });

const DEADLINE_MS = 15 * 60 * 1000;

const readPermissions = async (marker: string) => {
  try {
    const permissions = await Next.getForegroundPermissions();
    log(marker, 'permission read', permissions);
    return permissions;
  } catch (error) {
    log(marker, 'permission read THREW', String(error));
    return null;
  }
};

const runRd4 = async () => {
  log('4', 'BEGIN cached fix identity');
  try {
    const classic = await getLastKnownPositionAsync();
    log('4', 'classic getLastKnownPositionAsync', classicFixOf(classic));
    const next = await Next.getPosition({ timeout: 0 });
    log('4', 'next getPosition timeout 0', nextFixOf(next));
    if (classic === null || next === null) {
      log('4', 'END verdict', classic === null && next === null ? 'both null' : 'one null');
    } else {
      log('4', 'END verdict', classic.timestamp === next.timestamp ? 'same-timestamp fix' : 'different fixes');
    }
  } catch (error) {
    log('4', 'THREW', String(error));
    log('4', 'END');
  }
};

const runRd1 = async () => {
  log('1', 'BEGIN fresh stationary fix, phone flat on desk indoors');
  try {
    const fresh = await Next.getPosition({ maxCachedAge: 0, timeout: 30 });
    log('1', 'fresh fix', nextFixOf(fresh));
    log('1', 'END verdict', fresh === null ? 'no fresh fix within 30 s indoors' : `heading ${fresh.heading === null ? 'absent null' : 'present'}`);
  } catch (error) {
    log('1', 'THREW', String(error));
    log('1', 'END');
  }
};

const runRd5 = async (read: unknown) => {
  log('5', 'BEGIN coarse-only grant');
  log('5', 'permission read', read);
  try {
    const fix = await Next.getPosition({ timeout: 20 });
    log('5', 'getPosition resolved', nextFixOf(fix));
  } catch (error) {
    log('5', 'getPosition THREW', String(error));
  }
  log('5', 'END');
};

const runRd2 = async () => {
  log('2', 'BEGIN services off, calling enableLocationServices');
  try {
    const result = await Next.enableLocationServices();
    log('2', 'resolved', String(result));
    log('2', 'services after', String(Next.hasLocationServicesEnabled()));
  } catch (error) {
    log('2', 'THREW', String(error));
  }
  log('2', 'END');
};

const runAndroid = async () => {
  log('0', 'platform android begin');
  try {
    log('0', 'provider', String(await Next.getSelectedLocationProviderName()));
  } catch (error) {
    log('0', 'provider THREW', String(error));
  }
  log('0', 'services', String(Next.hasLocationServicesEnabled()));

  const initial = await readPermissions('0');
  if (initial?.granted) {
    await runRd4();
    if (Next.hasLocationServicesEnabled()) {
      await runRd1();
    } else {
      log('1', 'skipped services off');
    }
  } else {
    log('4', 'skipped permission not granted');
    log('1', 'skipped permission not granted');
  }

  let rd2Done = false;
  let rd5Done = false;
  let lastAccuracy = '';
  const deadline = Date.now() + DEADLINE_MS;
  while (Date.now() < deadline && !(rd2Done && rd5Done)) {
    if (!rd2Done && !Next.hasLocationServicesEnabled()) {
      rd2Done = true;
      await runRd2();
      continue;
    }
    if (!rd5Done) {
      const current = await readPermissions('5');
      const accuracy = String(current?.accuracy ?? 'unknown');
      if (accuracy !== lastAccuracy) {
        lastAccuracy = accuracy;
        log('5', 'perm-watch', current);
      }
      if (current?.accuracy === 'reduced') {
        rd5Done = true;
        await runRd5(current);
        continue;
      }
    }
    await wait(2000);
  }
  log('0', 'complete', { rd2: rd2Done, rd5: rd5Done });
};

const runIOS = async () => {
  log('0', 'platform ios begin');
  const initial = await readPermissions('0');
  if (initial?.granted) {
    await runRd4();
  } else {
    log('4', 'skipped permission not granted');
  }

  let rd3Done = false;
  let askedOnce = false;
  let lastAccuracy = '';
  const deadline = Date.now() + DEADLINE_MS;
  while (Date.now() < deadline && !rd3Done) {
    const current = await readPermissions('3');
    const accuracy = String(current?.accuracy ?? 'unknown');
    if (accuracy !== lastAccuracy) {
      lastAccuracy = accuracy;
      log('3', 'perm-watch', current);
    }
    if (current?.canAskAgain === true && current?.granted === false && !askedOnce) {
      askedOnce = true;
      log('3', 'not granted, requesting when-in-use plus full');
      try {
        log('3', 'initial resolved', await Next.requestForegroundPermissions({ accuracy: 'full' }));
      } catch (error) {
        log('3', 'initial THREW', String(error));
      }
      continue;
    }
    if (current?.granted === true && current?.accuracy === 'reduced') {
      rd3Done = true;
      log('3', 'BEGIN reduced grant, requesting full accuracy');
      try {
        log('3', 'resolved', await Next.requestForegroundPermissions({ accuracy: 'full' }));
        log('3', 'after-get', await Next.getForegroundPermissions());
      } catch (error) {
        log('3', 'THREW', String(error));
      }
      log('3', 'END');
      continue;
    }
    await wait(2000);
  }
  log('0', 'complete', { rd3: rd3Done });
};

export default function ProbeRoot() {
  const [note, setNote] = useState('probes running, host drives the choreography');
  const started = useRef(false);

  useEffect(() => {
    if (started.current) {
      return;
    }
    started.current = true;
    const flow = Platform.OS === 'android' ? runAndroid() : runIOS();
    flow.catch((error) => {
      log('0', 'FLOW THREW', String(error));
      setNote(`probes threw: ${String(error)}`);
    });
  }, []);

  return (
    <SafeAreaView style={styles.root}>
      <Text style={styles.text}>{note}</Text>
    </SafeAreaView>
  );
};

const styles = StyleSheet.create({
  root: { flex: 1, alignItems: 'center', justifyContent: 'center', backgroundColor: '#1b1626' },
  text: { color: '#ffffff' },
});
